   ```python
   import asyncio
   from typing import AsyncGenerator
   from anthropic import AsyncAnthropic

   class BaseAgent:
       def __init__(self, model: str = "claude-sonnet-4-6", system: str = ""):
           self.client = AsyncAnthropic()  # reads ANTHROPIC_API_KEY from env
           self.model = model
           self.system = system

       async def invoke(self, messages: list[dict]) -> str:
           """Send messages and return complete response. Retries 3x with exponential backoff."""
           for attempt in range(3):
               try:
                   response = await self.client.messages.create(
                       model=self.model,
                       max_tokens=4096,
                       system=self.system,
                       messages=messages,
                   )
                   return response.content[0].text
               except Exception:
                   if attempt == 2:
                       raise
                   await asyncio.sleep(2 ** attempt)
           return ""  # unreachable but satisfies type checker

       async def stream(self, messages: list[dict]) -> AsyncGenerator[str, None]:
           """Stream response tokens. Retries 3x with exponential backoff on connection failure."""
           for attempt in range(3):
               try:
                   async with self.client.messages.stream(
                       model=self.model,
                       max_tokens=4096,
                       system=self.system,
                       messages=messages,
                   ) as stream:
                       async for text in stream.text_stream:
                           yield text
                   return  # success — exit retry loop
               except Exception:
                   if attempt == 2:
                       raise
                   await asyncio.sleep(2 ** attempt)
   ```

   ```python
   from fastapi import APIRouter, Depends
   from fastapi.responses import EventSourceResponse
   from fastapi.sse import ServerSentEvent
   from sqlalchemy.ext.asyncio import AsyncSession
   from backend.db.base import get_session
   from backend.db.models import ChatSession, ChatMessage
   from backend.db.repository import BaseRepository
   from backend.schemas.chat import ChatMessageRequest
   from backend.agents.base_agent import BaseAgent
   import uuid
   from datetime import datetime, timezone

   router = APIRouter(tags=["chat"])

   AGENT_CONFIGS = {
       "facilitator": {"model": "claude-sonnet-4-6", "name": "ARIA", "system": "You are ARIA, a VFT (Value-Focused Thinking) facilitator. Help the user navigate the VFT process. Respond in Portuguese (PT-BR)."},
   }

   @router.post("/chat/message", response_class=EventSourceResponse)
   async def chat_message(
       body: ChatMessageRequest,
       session: AsyncSession = Depends(get_session),
   ):
       config = AGENT_CONFIGS.get(body.agent_type, AGENT_CONFIGS["facilitator"])
       agent = BaseAgent(model=config["model"], system=config["system"])

       # Persist user message
       msg_repo = BaseRepository(ChatMessage, session)

       # Get or create active session
       session_repo = BaseRepository(ChatSession, session)
       sessions = await session_repo.list(project_id=body.project_id, is_active=1)
       if sessions:
           chat_session = sessions[0]
       else:
           chat_session = await session_repo.create(
               project_id=body.project_id,
               is_active=1,
           )

       await msg_repo.create(
           session_id=chat_session.id,
           project_id=body.project_id,
           timestamp=datetime.now(timezone.utc).isoformat(),
           role="user",
           agent_name="User",
           content=body.message,
       )

       messages = [{"role": "user", "content": body.message}]

       async def generate():
           full_response = ""
           try:
               async for chunk in agent.stream(messages):
                   full_response += chunk
                   yield ServerSentEvent(data=chunk, event="delta")
           except Exception as e:
               yield ServerSentEvent(data=str(e), event="error")
               return

           # Persist assistant message
           await msg_repo.create(
               session_id=chat_session.id,
               project_id=body.project_id,
               timestamp=datetime.now(timezone.utc).isoformat(),
               role=body.agent_type,
               agent_name=config["name"],
               content=full_response,
           )
           await session.commit()

           yield ServerSentEvent(data="[DONE]", event="done")

       return EventSourceResponse(generate())
   ```

1. Create `frontend/src/api/chat.ts`:
   ```typescript
   const BASE_URL = import.meta.env.VITE_API_URL ?? window.location.origin

   export function streamChatMessage(
     message: string,
     projectId: string,
     agentType: string,
     onDelta: (text: string) => void,
     onDone: () => void,
     onError: (error: string) => void,
   ): () => void {
     const ctrl = new AbortController()

     fetch(`${BASE_URL}/api/chat/message`, {
       method: 'POST',
       headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
       body: JSON.stringify({ message, project_id: projectId, agent_type: agentType }),
       signal: ctrl.signal,
     }).then(async (res) => {
       if (!res.ok) {
         onError(`HTTP ${res.status}: ${res.statusText}`)
         return
       }
       const reader = res.body!.getReader()
       const decoder = new TextDecoder()
       let buf = ''
       while (true) {
         const { done, value } = await reader.read()
         if (done) { onDone(); break }
         buf += decoder.decode(value, { stream: true })
         const lines = buf.split('\n')
         buf = lines.pop() ?? ''
         for (const line of lines) {
           if (line.startsWith('data: ')) {
             const data = line.slice(6)
             if (data === '[DONE]') { onDone(); return }
             onDelta(data)
           }
           if (line.startsWith('event: error')) {
             // Next data line is the error
           }
         }
       }
     }).catch((err) => {
       if (err.name !== 'AbortError') {
         onError(err.message)
       }
     })

     return () => ctrl.abort()
   }
   ```

2. Create `frontend/src/hooks/useChat.ts`:
   ```typescript
   import { useCallback, useRef } from 'react'
   import { useChatStore } from '../stores/chatStore'
   import { useProjectStore } from '../stores/projectStore'
   import { streamChatMessage } from '../api/chat'
   import type { ChatMessage } from '../types'

   export function useChat() {
     const abortRef = useRef<(() => void) | null>(null)
     const { addMessage, appendToLastMessage, setStreaming, activeAgent } = useChatStore()
     const activeProjectId = useProjectStore((s) => s.activeProjectId)

     const sendMessage = useCallback((text: string) => {
       if (!activeProjectId || !text.trim()) return

       // Add user message
       const userMsg: ChatMessage = {
         id: crypto.randomUUID(),
         session_id: '',
         project_id: activeProjectId,
         timestamp: new Date().toISOString(),
         role: 'user',
         agent_name: 'User',
         content: text,
         phase: null,
         step: null,
       }
       addMessage(userMsg)

       // Add placeholder for assistant response
       const assistantMsg: ChatMessage = {
         id: crypto.randomUUID(),
         session_id: '',
         project_id: activeProjectId,
         timestamp: new Date().toISOString(),
         role: 'assistant',
         agent_name: activeAgent,
         content: '',
         phase: null,
         step: null,
       }
       addMessage(assistantMsg)
       setStreaming(true)

       abortRef.current = streamChatMessage(
         text,
         activeProjectId,
         'facilitator',
         (delta) => appendToLastMessage(delta),
         () => setStreaming(false),
         (error) => {
           appendToLastMessage(`\n\n[Erro: ${error}]`)
           setStreaming(false)
         },
       )
     }, [activeProjectId, activeAgent, addMessage, appendToLastMessage, setStreaming])

     const cancelStream = useCallback(() => {
       abortRef.current?.()
       setStreaming(false)
     }, [setStreaming])

     return { sendMessage, cancelStream }
   }
   ```
