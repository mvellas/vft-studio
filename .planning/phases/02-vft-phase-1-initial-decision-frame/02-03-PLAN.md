       ```typescript
       import { apiClient } from './client'

       export interface IDFData {
         idf_purpose: string
         idf_perspective: string
         idf_scope: string
         idf_documented_at: string | null
       }

       export interface IDFUpdate {
         idf_purpose?: string
         idf_perspective?: string
         idf_scope?: string
       }

       export const decisionFrameApi = {
         get: (projectId: string) =>
           apiClient.get<IDFData>(`/api/projects/${projectId}/decision-frame`).then(r => r.data),
         save: (projectId: string, data: IDFUpdate) =>
           apiClient.put(`/api/projects/${projectId}/decision-frame`, data).then(r => r.data),
         generateSuggestion: (projectId: string, field: string, hint: string = '') =>
           apiClient.post<{ suggestion: string }>(`/api/projects/${projectId}/frame/validate-idf`, {
             purpose: field === 'purpose' ? hint : '',
             perspective: field === 'perspective' ? hint : '',
             scope: field === 'scope' ? hint : '',
           }).then(r => r.data),
       }
       ```
